cmake_minimum_required(VERSION 3.16)
project(vehicle_sensor_monitor LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(REQUIRE_IPC_TESTS "Fail rather than skip when Unix sockets are sandboxed" OFF)
option(ENABLE_TSAN "Enable ThreadSanitizer" OFF)
find_package(Threads REQUIRED)
add_library(sensor_core src/Sensor.cpp src/Watchdog.cpp src/Shutdown.cpp src/Transport.cpp)
target_include_directories(sensor_core PUBLIC src)
target_link_libraries(sensor_core PUBLIC Threads::Threads)
target_compile_options(sensor_core PRIVATE -Wall -Wextra -Wpedantic)
if(ENABLE_TSAN)
  if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(sensor_core PUBLIC -fsanitize=thread -fno-omit-frame-pointer -g)
    target_link_options(sensor_core PUBLIC -fsanitize=thread)
  else()
    message(FATAL_ERROR "ThreadSanitizer requires Clang or GCC")
  endif()
endif()
add_executable(sensor_monitor src/main.cpp)
target_link_libraries(sensor_monitor PRIVATE sensor_core)
add_executable(sensor_simulator src/simulator_main.cpp)
target_link_libraries(sensor_simulator PRIVATE sensor_core)
enable_testing()
add_executable(test_watchdog tests/test_watchdog.cpp)
target_link_libraries(test_watchdog PRIVATE sensor_core)
add_test(NAME watchdog COMMAND test_watchdog)
set_tests_properties(watchdog PROPERTIES TIMEOUT 5)

add_executable(test_transport tests/test_transport.cpp)
target_link_libraries(test_transport PRIVATE sensor_core)
set(IPC_TEST_ARGS)
if(REQUIRE_IPC_TESTS)
  list(APPEND IPC_TEST_ARGS --require-ipc)
endif()
add_test(NAME protocol COMMAND test_transport --protocol)
add_test(NAME transport COMMAND test_transport ${IPC_TEST_ARGS})
set_tests_properties(transport PROPERTIES TIMEOUT 5 SKIP_RETURN_CODE 77)
find_package(Python3 COMPONENTS Interpreter QUIET)
if(REQUIRE_IPC_TESTS AND NOT Python3_Interpreter_FOUND)
  message(FATAL_ERROR "Python 3 is required for mandatory IPC process tests")
endif()
if(Python3_Interpreter_FOUND)
  add_test(NAME process_signals COMMAND ${Python3_EXECUTABLE}
    ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_processes.py
    $<TARGET_FILE:sensor_monitor> $<TARGET_FILE:sensor_simulator> --signals)
  set_tests_properties(process_signals PROPERTIES TIMEOUT 30)
  add_test(NAME process_integration COMMAND ${Python3_EXECUTABLE}
    ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_processes.py
    $<TARGET_FILE:sensor_monitor> $<TARGET_FILE:sensor_simulator> ${IPC_TEST_ARGS})
  set_tests_properties(process_integration PROPERTIES TIMEOUT 45 SKIP_RETURN_CODE 77)
endif()
